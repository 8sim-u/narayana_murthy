# Vision pipeline dependencies for the edge node (Person A).
#
# These are intentionally NOT in backend/requirements.txt - the backend runs
# headless on a server, while this pipeline runs on a machine with cameras.
#
# Install with:  pip install -r vision/requirements.txt
#
# NOTE: ultralytics also pulls torch and opencv-python transitively, so this
# install is several GB even though they are listed explicitly. It has NOT
# been verified against a live camera feed yet - treat the constraints below
# as a starting point, not a lockfile.
# The YOLO weights (yolov8n.pt) download automatically on first run.

ultralytics>=8.4,<9
opencv-python>=4.10,<6
torch>=2.0,<3
requests>=2.32,<3

# OPTIONAL: needed only for WAMSS_PRIVACY_MODE=1 (privacy-preserving presence
# mode in vision/multicamera.py). Provides the on-device MediaPipe backend
# (face detection for person counting, pose for the coarse standing/sitting
# aggregate) and face blurring for anonymized occupancy snapshots.
#
# If absent, privacy mode still works via a graceful fallback: the YOLO
# person-count aggregate path (no pose, no track ids), and ANY frame that
# would need storing is refused with a logged reason instead. The
# non-privacy path never needs MediaPipe.
#
# Install explicitly on the edge node when privacy mode is required:
#   pip install mediapipe
